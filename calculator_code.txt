  <script>
    const resultEl = document.getElementById('result');
    const expressionEl = document.getElementById('expression');

    let current = '0';
    let previous = null;
    let operator = null;
    let shouldReset = false;
    let expressionStr = '';

    function updateDisplay() {
      resultEl.textContent = current;
      expressionEl.textContent = expressionStr;
      const len = current.length;
      resultEl.className = 'result' + (len > 12 ? ' xsmall' : len > 9 ? ' small' : '');
    }

    function inputNum(val) {
      if (shouldReset) {
        current = val === '0' ? '0' : val;
        shouldReset = false;
      } else {
        if (current === '0' && val !== '.') current = val;
        else if (current.length < 16) current += val;
      }
      updateDisplay();
    }

    function inputDecimal() {
      if (shouldReset) { current = '0.'; shouldReset = false; updateDisplay(); return; }
      if (!current.includes('.')) { current += '.'; updateDisplay(); }
    }

    function inputOperator(op) {
      const val = parseFloat(current);
      if (previous !== null && !shouldReset) {
        const res = compute(previous, val, operator);
        current = format(res);
        previous = res;
      } else {
        previous = val;
      }
      operator = op;
      expressionStr = format(previous) + ' ' + op;
      shouldReset = true;
      updateDisplay();
    }

    function compute(a, b, op) {
      switch (op) {
        case '+': return a + b;
        case '−': return a - b;
        case '×': return a * b;
        case '÷': return b === 0 ? NaN : a / b;
      }
    }

    function format(n) {
      if (!isFinite(n)) return 'Error';
      // Avoid floating-point display noise
      const s = parseFloat(n.toPrecision(12)).toString();
      return s;
    }

    function equals() {
      if (operator === null || previous === null) return;
      const val = parseFloat(current);
      const res = compute(previous, val, operator);
      expressionStr = format(previous) + ' ' + operator + ' ' + format(val) + ' =';
      current = format(res);
      previous = null;
      operator = null;
      shouldReset = true;
      updateDisplay();
    }

    function clear() {
      current = '0'; previous = null; operator = null;
      shouldReset = false; expressionStr = '';
      updateDisplay();
    }

    function sign() {
      if (current === '0') return;
      current = current.startsWith('-') ? current.slice(1) : '-' + current;
      updateDisplay();
    }

    function percent() {
      current = format(parseFloat(current) / 100);
      updateDisplay();
    }

    // Button clicks
    document.querySelector('.buttons').addEventListener('click', e => {
      const btn = e.target.closest('button');
      if (!btn) return;
      const { action, val, op } = btn.dataset;
      if (action === 'num')     inputNum(val);
      if (action === 'decimal') inputDecimal();
      if (action === 'op')      inputOperator(op);
      if (action === 'equals')  equals();
      if (action === 'clear')   clear();
      if (action === 'sign')    sign();
      if (action === 'percent') percent();
    });

    // Keyboard support
    document.addEventListener('keydown', e => {
      if (e.key >= '0' && e.key <= '9') inputNum(e.key);
      else if (e.key === '.') inputDecimal();
      else if (e.key === '+') inputOperator('+');
      else if (e.key === '-') inputOperator('−');
      else if (e.key === '*') inputOperator('×');
      else if (e.key === '/') { e.preventDefault(); inputOperator('÷'); }
      else if (e.key === 'Enter' || e.key === '=') equals();
      else if (e.key === 'Escape') clear();
      else if (e.key === 'Backspace') {
        if (shouldReset) return;
        current = current.length > 1 ? current.slice(0, -1) : '0';
        updateDisplay();
      }
      else if (e.key === '%') percent();
    });

    updateDisplay();
  </script>
